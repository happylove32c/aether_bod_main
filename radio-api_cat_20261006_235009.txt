===== index.js =====
const express = require('express')
const { MongoClient } = require('mongodb')
require('dotenv').config()

const app = express()
app.use(express.json())

const client = new MongoClient(process.env.MONGODB_URI)
let stations

// --- Routes ---
app.get('/stations', async (req, res, next) => {
  try {
    const data = await stations.find({}).toArray()
    res.json(data)
  } catch (err) {
    next(err)
  }
})


app.post('/stations', async (req, res, next) => {
  try {
    const { frequency, name, url, genre } = req.body

    const result = await stations.insertOne({
      frequency,
      name,
      url,
      genre: genre || null,
      createdAt: new Date()
    })

    res.status(201).json({ message: 'station added', id: result.insertedId })
  } catch (err) {
    next(err)
  }
})

// --- Startup ---
async function start() {
  try {
    await client.connect()
    const db = client.db('radio')
    stations = db.collection('stations')
    console.log('Connected to MongoDB')

    const PORT = process.env.PORT || 3000
    const server = app.listen(PORT, () => {
      console.log(`Server running on port ${PORT}`)
    })

  } catch (err) {
    console.error('Failed to start server:', err)
    process.exit(1)
  }
}
